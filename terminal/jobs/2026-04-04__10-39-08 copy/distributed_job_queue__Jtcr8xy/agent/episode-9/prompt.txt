import requests
import redis
import schedule
import time
import os

API_URL = os.environ.get("API_URL", "http://localhost:8000")
redis_client = redis.from_url(os.environ.get("REDIS_URL", "redis://localhost:6379"))

def submit_scheduled_jobs():
    payloads = ["health_check", "cleanup_old_records", "generate_report"]
    for i, payload in enumerate(payloads):
        try:
            # Bug 5: priority is str not int
            r = requests.post(f"{API_URL}/jobs", json={"payload": payload, "priority": str(i)}, timeout=5)
            print(f"Scheduled {payload}: {r.status_code} {r.text}")
        except Exception as e:
            print(f"Scheduler error: {e}")

def check_queue_depth():
    depth = redis_client.llen("job_queue")
    # Bug 6: stores as string — should be int
    redis_client.set("queue_depth", str(depth))

schedule.every(30).seconds.do(submit_scheduled_jobs)
schedule.every(10).seconds.do(check_queue_depth)

if __name__ == "__main__":
    time.sleep(8)
    print("Scheduler started")
    submit_scheduled_jobs()
    while True:
        schedule.run_pending()
        time.sleep(1)
