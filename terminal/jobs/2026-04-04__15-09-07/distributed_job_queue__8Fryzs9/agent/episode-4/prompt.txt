from flask import Flask, jsonify, request
from sqlalchemy import create_engine, text
import redis
import os
import json
import time

app = Flask(__name__)

# Bug 1: wrong env var — should be DATABASE_URL
engine = create_engine(os.environ.get("DB_URL", "sqlite:////tmp/taskdb.sqlite"))
redis_client = redis.from_url(os.environ.get("REDIS_URL", "redis://localhost:6379"))

def init_db():
    with engine.connect() as conn:
        conn.execute(text("""
            CREATE TABLE IF NOT EXISTS jobs (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                payload TEXT NOT NULL,
                status TEXT DEFAULT 'pending',
                priority INTEGER DEFAULT 0,
                created_at TEXT DEFAULT (datetime('now')),
                processed_at TEXT
            )
        """))
        conn.commit()

@app.route("/health")
def health():
    try:
        with engine.connect() as conn:
            conn.execute(text("SELECT 1"))
        redis_client.ping()
        return jsonify({"status": "healthy", "db": "ok", "redis": "ok"})
    except Exception as e:
        return jsonify({"status": "unhealthy", "error": str(e)}), 500

@app.route("/jobs", methods=["POST"])
def create_job():
    data = request.get_json()
    if not data or "payload" not in data:
        return jsonify({"error": "payload required"}), 400
    priority = data.get("priority", 0)
    if not isinstance(priority, int):
        return jsonify({"error": "priority must be integer"}), 400
    with engine.connect() as conn:
        result = conn.execute(
            text("INSERT INTO jobs (payload, priority) VALUES (:payload, :priority)"),
            {"payload": data["payload"], "priority": priority}
        )
        conn.commit()
        job_id = result.lastrowid
    # Bug 2: wrong queue name
    redis_client.lpush("jobs", json.dumps({"id": job_id, "payload": data["payload"], "priority": priority}))
    return jsonify({"id": job_id, "status": "pending"}), 201

@app.route("/jobs/<int:job_id>")
def get_job(job_id):
    with engine.connect() as conn:
        result = conn.execute(
            text("SELECT id, payload, status, priority, created_at FROM jobs WHERE id = :id"),
            {"id": job_id}
        )
        row = result.fetchone()
    if not row:
        return jsonify({"error": "not found"}), 404
    return jsonify({"id": row[0], "payload": row[1], "status": row[2], "priority": row[3], "created_at": str(row[4])})

@app.route("/jobs")
def list_jobs():
    status = request.args.get("status")
    with engine.connect() as conn:
        if status:
            # Bug 3: ASC should be DESC
            result = conn.execute(
                text("SELECT id, payload, status, priority FROM jobs WHERE status = :status ORDER BY priority ASC"),
                {"status": status}
            )
        else:
            result = conn.execute(text("SELECT id, payload, status, priority FROM jobs ORDER BY created_at DESC"))
        rows = result.fetchall()
    return jsonify([{"id": r[0], "payload": r[1], "status": r[2], "priority": r[3]} for r in rows])

@app.route("/stats")
def stats():
    with engine.connect() as conn:
        result = conn.execute(text("SELECT status, COUNT(*) FROM jobs GROUP BY status"))
        rows = result.fetchall()
    return jsonify({row[0]: row[1] for row in rows})

if __name__ == "__main__":
    init_db()
