import redis
import json
import os
import time
from sqlalchemy import create_engine, text

redis_client = redis.from_url(os.environ.get("REDIS_URL", "redis://localhost:6379"))
engine = create_engine(os.environ.get("DATABASE_URL", "sqlite:////tmp/taskdb.sqlite"))

def process_job(job_data):
    job_id = job_data["id"]
    time.sleep(0.1)
    with engine.connect() as conn:
        # Bug 4: wrong status — should be 'completed'
        conn.execute(
            text("UPDATE jobs SET status = 'done', processed_at = datetime('now') WHERE id = :id"),
            {"id": job_id}
        )
        conn.commit()
    print(f"Job {job_id} processed")

def main():
    print("Worker listening on job_queue...")
    while True:
        try:
            result = redis_client.brpop("job_queue", timeout=5)
            if result:
                _, data = result
                process_job(json.loads(data))
        except Exception as e:
            print(f"Worker error: {e}")
            time.sleep(1)

if __name__ == "__main__":
    main()
